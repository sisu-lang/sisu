# The neighbouring forms of the checker's special cases: each spelling of a value that the
# checker treats in its own way, and each position that a join or an expected type gives it.
#
# A fix brief and a review take their "fix the class" list from here: list each value in each
# context, and probe each pair. When a review finds a neighbour that this file does not name, add
# it here, so that the next list has it.
#
# `tests/neighbours.rs` puts each value of `[values]` into the `$` of each line of `[contexts]`,
# writes the statement into the two `{stmt}` places of `[template]` (`main` and a generic
# function), and runs `sisu check` on each program. It also writes it into the one `{stmt}` of
# `[failed]`, a function whose bound failed, in a second program: the template with no statement,
# then that function. That program always fails, so `check` stops before the later passes. The
# first program can reach them. Most programs are wrong. The test fails when `sisu check` crashes
# on one, as a reached `panic!` or `assert!` in the checker does.
#
# A line that starts with `#` is a comment. Each value and each context is one line.

[template]
interface Named {
    fn name(self) -> string
}
interface Pet: Named {
    fn tail(self) -> i64
}
class Cat: Named {
    fn name(self) -> string { "cat" }
}
class Fox: Pet {
    fn name(self) -> string { "fox" }
    fn tail(self) -> i64 { 1 }
}
class Rock {}
enum E {
    A
    fn push(self, x: i64) -> i64 { x }
}
class Bx {
    fn wrap<T>(self, v: T) -> T { v }
    fn none<T>(self) -> T? { None }
    fn put<T>(self, mut x: T?, n: i64) { x = None }
}
type M<T> = T?
type MM<T> = T??
type OP = (any Pet)??
class Bq<T> {
    v: T
}
type Bqa<T> = Bq<T>
fn nothing<T>() -> T? { None }
fn nn<T>() -> T?? { None }
fn n3<T>() -> T??? { None }
class Oa {
    oa: [i64]?
}
fn oa_none() -> Oa? { None }
fn mk<T>() -> T { loop {} }
fn some<T>(v: T) -> T? { Some(v) }
fn id<T>(v: T) -> T { v }
fn take<T>(mut x: T?) { x = None }
fn take2<T>(mut x: T?, n: i64) { x = None }
fn keep<T>(mut x: T?) -> T? { x }
fn opt<T>(v: T?) -> T? { v }
fn generic<U: Named, V>(u: U, v: V, c: bool, d: bool, k: i64, n: any Named, p: any Pet, op: (any Pet)?, ob: Bx?) {
    let mut mq: (any Pet)? = op
    {stmt}
}
fn main() {
    let c = true
    let d = false
    let k = 1
    let n: any Named = Cat()
    let p: any Pet = Fox()
    let op: (any Pet)? = Fox()
    let ob: Bx? = Bx()
    generic(Cat(), Rock(), c, d, k, n, p, op, ob)
    let mut mq: (any Pet)? = op
    {stmt}
}

[failed]
fn failed<U: Missing, V>(u: U, v: V, c: bool, d: bool, k: i64, n: any Named, p: any Pet, op: (any Pet)?, ob: Bx?) {
    let mut mq: (any Pet)? = op
    {stmt}
}
